% Résumé template for LuaLaTeX, expanded by assets.RenderResume.
%
% Built on moderncv with its banking style, the sans option and a text area
% of 80 percent of the page in each direction. The title block and the entry
% layout are written here rather than taken from the class: the class's
% contact line draws Font Awesome icons, whose glyph names would end up in
% the text layer, and its \cventry boxes an entry's description so a long
% entry could not break across pages.
%
% Every printed value goes through one of the encoders tex, rich, url or
% paper; an empty optional field emits no command at all.
\documentclass[11pt,<% paper .Paper %>,sans]{moderncv}
\moderncvstyle{banking}
\moderncvcolor{blue}
% One accent colour, a deep blue, for the name, the section headings and the
% bullet markers; everything else is black or the class's dark grey.
\definecolor{accent}{RGB}{28,61,105}
\colorlet{firstnamecolor}{accent}
\colorlet{lastnamecolor}{accent}
\colorlet{sectioncolor}{accent}
\colorlet{bodyrulecolor}{accent}
\usepackage[scale=0.80]{geometry}
\nopagenumbers
% A long unbreakable word (a path, a command name) may loosen its line rather
% than run into the right margin.
\setlength{\emergencystretch}{3em}
\makeatletter

% The class's Latin Modern Sans, declared here with the kernel's file names
% and optical sizes but with kerning off: pypdf reads a wide kern (A-W in
% "AWS") as a word break. Declaring the family keeps the kernel from loading
% its own definitions over these. Nothing else about the fonts changes.
\newcommand*{\unkerned}[1]{"[#1]:+tlig;-kern;"}
\DeclareFontFamily{TU}{lmss}{}
\DeclareFontShape{TU}{lmss}{m}{n}{<-8.5>\unkerned{lmsans8-regular}
  <8.5-9.5>\unkerned{lmsans9-regular}<9.5-11>\unkerned{lmsans10-regular}
  <11-15.5>\unkerned{lmsans12-regular}<15.5->\unkerned{lmsans17-regular}}{}
\DeclareFontShape{TU}{lmss}{m}{sl}{<-8.5>\unkerned{lmsans8-oblique}
  <8.5-9.5>\unkerned{lmsans9-oblique}<9.5-11>\unkerned{lmsans10-oblique}
  <11-15.5>\unkerned{lmsans12-oblique}<15.5->\unkerned{lmsans17-oblique}}{}
\DeclareFontShape{TU}{lmss}{bx}{n}{<->\unkerned{lmsans10-bold}}{}
\DeclareFontShape{TU}{lmss}{bx}{sl}{<->\unkerned{lmsans10-boldoblique}}{}
\DeclareFontShape{TU}{lmss}{m}{it}{<->ssub*lmss/m/sl}{}
\DeclareFontShape{TU}{lmss}{bx}{it}{<->ssub*lmss/bx/sl}{}
\DeclareFontShape{TU}{lmss}{b}{n}{<->ssub*lmss/bx/n}{}
\DeclareFontShape{TU}{lmss}{b}{sl}{<->ssub*lmss/bx/sl}{}
\DeclareFontShape{TU}{lmss}{b}{it}{<->ssub*lmss/bx/sl}{}

\name{<% tex .FirstName %>}{<% tex .LastName %>}
% The class builds the PDF title from the name at the end of the preamble;
% this hook runs after it, so the document title wins.
\AtEndPreamble{\hypersetup{pdftitle={<% tex .Title %>}}}

% Spacing. The two modes differ only in these three values: the space after
% an entry, the space above a list (as the list's top separation) and the
% space between list items. Paragraphs inside an item never get extra space.
<% if .Tight %>\newcommand*{\entryafterspace}{5pt}
\newcommand*{\setlistspacing}{\setlength{\topsep}{0pt}\setlength{\itemsep}{0pt}}
<% else %>\newcommand*{\entryafterspace}{7pt}
\newcommand*{\setlistspacing}{\setlength{\topsep}{2pt}\setlength{\itemsep}{2pt}}
<% end %>
% Contact line: items joined by a bar that disappears where the line wraps,
% so no line starts or ends with a separator (the \unskip removes the space
% the source's line end leaves before an item).
\newif\ifcontactbegun
\newcommand*{\contactitem}[1]{%
  \ifcontactbegun\unskip\discretionary{}{}{\kern.6em\textbar\kern.6em}\fi
  \contactbeguntrue\mbox{#1}}

% Entries. The fields are set one by one, so an empty field is simply never
% set; \entryhead prints title and organization on the left, wrapping in
% the width the right column leaves, and location and dates on the right,
% level with the first line; the note goes on a line of its own below.
%
% Keeping things together. The class's \section forbids a break after the
% heading; the header and note lines cannot be broken apart; a bullet list
% forbids a break before its first item, and a bullet splits only with two
% lines on each side. So a heading always shares its page with the first
% entry's header, and that header with the start of its first bullet:
% exactly the space needed is reserved. Every other break (between items,
% after a list, between paragraphs) is allowed, and with the class's ragged
% bottom TeX takes the last one that fits, so pages fill.
\newcommand*{\entrydates}[1]{\def\entry@dates{#1}}
\newcommand*{\entrytitle}[1]{\def\entry@title{#1}}
\newcommand*{\entryorg}[1]{\def\entry@org{#1}}
\newcommand*{\entryplace}[1]{\def\entry@place{#1}}
\newcommand*{\entrynote}[1]{\def\entry@note{#1}}
\newcommand*{\entry@reset}{%
  \let\entry@dates\@empty\let\entry@title\@empty\let\entry@org\@empty
  \let\entry@place\@empty\let\entry@note\@empty}
\newenvironment{entry}
  {\par\entry@reset}
  {\par\addvspace{\entryafterspace}}
\newsavebox{\entry@right}
\newcommand*{\entryhead}{%
  \begingroup\interlinepenalty\@M
  \sbox{\entry@right}{%
    \ifx\entry@place\@empty\else
      \entry@place\ifx\entry@dates\@empty\else\hspace{.5em}\textbar\hspace{.5em}\fi\fi
    \ifx\entry@dates\@empty\else\entry@dates\fi}%
  \noindent\parbox[t]{\dimexpr\linewidth-\wd\entry@right-1.5em\relax}{%
    \raggedright
    \ifx\entry@title\@empty\else
      {\bfseries\entry@title\ifx\entry@org\@empty\else,\fi}\fi
    \ifx\entry@org\@empty\else
      \ifx\entry@title\@empty\else\space\fi{\itshape\entry@org}\fi\strut}%
  \hfill\usebox{\entry@right}\par
  \ifx\entry@note\@empty\else\nobreak{\itshape\entry@note\par}\fi
  \endgroup}

% Bullets: a plain list in the body font, outside any box so that it can
% break across pages. The marker is the bullet of the sans font itself.
\newenvironment{bullets}
  {\begin{list}{\textcolor{accent}{\textbullet}}{%
     \setlength{\leftmargin}{1.1em}\setlength{\labelsep}{.5em}%
     \setlength{\labelwidth}{.6em}\setlength{\partopsep}{0pt}%
     \setlength{\parsep}{0pt}\setlistspacing
     \@beginparpenalty\@M\@itempenalty\z@\@endparpenalty\z@
     \clubpenalty\@M\widowpenalty\@M}}
  {\end{list}}

\makeatother
